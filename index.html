How this stays in sync

Entry → Sheet → every device

Every time you tap Add & sync entry, it's sent straight to your Google Sheet — from laptop or mobile, doesn't matter which. If you're offline, it's held here as pending and retried next time you hit refresh or add another entry.

The Ledger and Dashboard tabs always read the full history straight back from the Sheet — that's the single source of truth, not any one browser. So whatever device you check from, you'll see the same complete picture, no more manually copying calendar notes into Excel at month-end.

First time here? Go to the Settings tab and paste in your Apps Script Web App URL, and set your milk price per liter.

Pricing

Price per liter history

Each month is billed at the latest price whose "effective from" month is on or before it. When milk price goes up, add the new price with its starting month — older months keep their old price. Prices are stored in a "Prices" tab of your Google Sheet, so any new device or browser picks them up automatically once the Sheet URL is set.
One-time setup

Google Apps Script backend (paste this in once)

1. Create a Google Sheet. 2. Extensions → Apps Script. 3. Paste the code below. 4. Deploy → New deployment → Web app → Execute as "Me" → Who has access "Anyone". 5. Copy the Web App URL into the field above.

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var body = JSON.parse(e.postData.contents);

  if (body && body.action === 'setPrices') {
    var ps = ss.getSheetByName('Prices') || ss.insertSheet('Prices');
    ps.clear();
    ps.getRange('A:A').setNumberFormat('@');   // keep "2026-10" as text, not a date
    ps.getRange(1, 1, 1, 2).setValues([['from', 'price']]);
    var list = body.prices || [];
    if (list.length) {
      ps.getRange(2, 1, list.length, 2)
        .setValues(list.map(function(p){ return [String(p.from), Number(p.price)]; }));
    }
    return ContentService.createTextOutput(JSON.stringify({status:'ok'}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var sheet = ss.getSheetByName('Entries') || ss.insertSheet('Entries');
  body.forEach(function(r) {
    sheet.appendRow([r.date, r.morning, r.evening, r.note]);
  });
  return ContentService.createTextOutput(JSON.stringify({status:'ok'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tz = ss.getSpreadsheetTimeZone();
  var rows = [];

  if (e.parameter.action === 'prices') {
    var ps = ss.getSheetByName('Prices');
    if (ps && ps.getLastRow() > 1) {
      rows = ps.getRange(2, 1, ps.getLastRow() - 1, 2).getValues().map(function(r) {
        var f = r[0];
        if (Object.prototype.toString.call(f) === '[object Date]') {
          f = Utilities.formatDate(f, tz, 'yyyy-MM');
        }
        return { from: String(f).slice(0, 7), price: r[1] };
      }).filter(function(r){ return r.from && r.price; });
    }
  } else {
    var sheet = ss.getSheetByName('Entries');
    if (sheet) {
      rows = sheet.getDataRange().getValues().map(function(r) {
        var dateStr = r[0];
        if (Object.prototype.toString.call(dateStr) === '[object Date]') {
          dateStr = Utilities.formatDate(dateStr, tz, 'yyyy-MM-dd');
        }
        return { date: dateStr, morning: r[1], evening: r[2], note: r[3] };
      }).filter(function(r){ return r.date; });
    }
  }

  var json = JSON.stringify(rows);
  if (e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + '(' + json + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
Note: reading data back uses JSONP (a script-tag fetch) rather than a normal request — this is because Google's Web App doesn't send the browser permission headers a regular fetch needs to read a response from a different site. JSONP works around that reliably; you don't need to do anything differently, it's handled automatically. The date is explicitly formatted using the Sheet's own timezone to avoid a day-shift bug that occurs when Sheets auto-converts a date string into a real Date cell.
